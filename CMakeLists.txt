cmake_minimum_required(VERSION 3.16)
project(fxfighter C)
if(WIN32)
  enable_language(RC)
endif()

set(CMAKE_C_STANDARD 11)
find_package(SDL2 REQUIRED)
find_package(OpenGL REQUIRED)

add_executable(fxfighter WIN32 src/main.c src/assets.c src/fight.c src/audio.c src/ui.c src/fe3d.c src/fe3d_gl.c src/fe3d_vm.c src/video.c src/sprfx.c)
target_include_directories(fxfighter PRIVATE ${SDL2_INCLUDE_DIRS})
if(TARGET SDL2::SDL2)
  target_link_libraries(fxfighter PRIVATE SDL2::SDL2main SDL2::SDL2 OpenGL::GL)
else()
  target_link_libraries(fxfighter PRIVATE ${SDL2_LIBRARIES} OpenGL::GL)
endif()
if(NOT MSVC)
  target_link_libraries(fxfighter PRIVATE m)
  target_compile_options(fxfighter PRIVATE -Wall -Wextra -O2)
endif()

# headless checks (no SDL): fight simulator, the 3D front end asset check and its script VM
add_executable(fightsim tests/fightsim.c src/fight.c src/assets.c)
add_executable(fecheck tests/fecheck.c src/fe3d.c src/assets.c)
add_executable(fevm tests/fevm.c src/fe3d_vm.c src/fe3d.c src/assets.c)
add_executable(brpcheck tests/brpcheck.c src/video.c)
add_executable(audiocheck tests/audiocheck.c src/audio.c src/assets.c)
target_include_directories(audiocheck PRIVATE ${SDL2_INCLUDE_DIRS} src)
if(TARGET SDL2::SDL2)
  target_link_libraries(audiocheck PRIVATE SDL2::SDL2main SDL2::SDL2)
else()
  target_link_libraries(audiocheck PRIVATE ${SDL2_LIBRARIES})
endif()
if(NOT MSVC)
  target_link_libraries(fightsim PRIVATE m)
  target_link_libraries(fecheck PRIVATE m)
  target_link_libraries(fevm PRIVATE m)
  target_link_libraries(audiocheck PRIVATE m)
endif()

# Windows launcher: edits options.cfg with ordinary controls (resolution, volumes, rules, keys, detail) and starts the game
if(WIN32)
  if(MSVC)
    add_executable(fxfighter-launcher WIN32 launcher/launcher.c launcher/launcher.rc)
  else()
    add_executable(fxfighter-launcher WIN32 launcher/launcher.c launcher/launcher.rc)
    target_link_options(fxfighter-launcher PRIVATE -municode -static -static-libgcc)
  endif()
  target_compile_definitions(fxfighter-launcher PRIVATE UNICODE _UNICODE)
  target_link_libraries(fxfighter-launcher PRIVATE comctl32 comdlg32 shell32 ole32)
endif()
